#!/bin/bash
set -euo pipefail
exec >/root/gpu-baseline.log 2>&1
trap 'echo GPU_BASELINE_FAILED line=$LINENO' ERR
export LD_LIBRARY_PATH=/usr/local/cuda-12.9/targets/x86_64-linux/lib:${LD_LIBRARY_PATH:-}
export VOLTA_GRAPHS=4
model=/root/model/DeepSeek-V4.1-Flash-1017-Volta.gguf
while [ ! -f /root/MODEL_READY ]; do
 [ "$(date +%s)" -lt 1792905506 ] || exit 2
 sleep 10
done
[ -f /root/BOOTSTRAP_DONE ]
[ -f /root/eval.done ]
date -u
cd /root/volta
./target/release/volta eval --model "$model" --eval /root/eval --seqs 8 --len 1024 --chunk 1 >/root/teacher-eval.jsonl 2>/root/teacher-eval.log
python3 - <<'PY'
import json,math,pathlib
rows=[json.loads(l) for l in pathlib.Path('/root/teacher-eval.jsonl').read_text().splitlines() if l.startswith('{')];x=rows[-1];assert x['tokens']==8184,x
assert all(math.isfinite(x[k]) for k in ['ppl','kl_top256','top1_agree']),x
assert x['ppl']<=10.5 and x['kl_top256']<=.2 and x['top1_agree']>=.8,x
pathlib.Path('/root/TEACHER_GATE_PASSED').write_text(json.dumps(x,indent=2))
PY
nvidia-smi --query-gpu=index,name,memory.used,clocks.sm,clocks.mem,power.limit --format=csv >/root/pre-server-gpus.csv
exec ./target/release/volta-server --model "$model" --gpus 8 --slots 1 --ctx 131072 --chunk 64 --spec-k 5 --temperature 0.7 --top-p 1 --max-tokens 16384 --host 127.0.0.1 --port 8090 --admin-port 8091 >/root/server-baseline.log 2>&1
