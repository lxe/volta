#!/usr/bin/env python3
"""Run a bounded teacher quality check after both owned shape gauntlets finish."""
import argparse
import fcntl
import json
import pathlib
import re
import subprocess
import time
import urllib.request

ROOT = pathlib.Path(__file__).parent
SSH = ['ssh', '-F', str(ROOT / 'ssh.conf'), 'volta-run']
ap = argparse.ArgumentParser(description=__doc__)
ap.add_argument('--heads', action='store_true')
ap.add_argument('--name', required=True)
a = ap.parse_args()
assert re.fullmatch(r'[a-zA-Z0-9_-]+', a.name)
assert (ROOT / 'optimized-gauntlet/GAUNTLET_DONE').exists()
assert time.time() < 1792907006
lock = (ROOT / 'server-transition.lock').open('w')
fcntl.flock(lock, fcntl.LOCK_EX | fcntl.LOCK_NB)
out = ROOT / 'teacher-runs' / a.name
out.mkdir(parents=True, exist_ok=False)
with urllib.request.urlopen('http://127.0.0.1:8091/stats', timeout=10) as f:
    old = json.load(f)
assert old['active'] == 0 and old['queued'] == 0
(out / 'previous-stats.json').write_text(json.dumps(old, indent=2))
code = '''import os,pathlib,json,signal,time,subprocess,hashlib
cfg=json.loads(pathlib.Path('/root/current-server.json').read_text())
allowed={'/root/volta/target/release/volta-server','/root/volta-candidate/target/release/volta-server','/root/volta-graph-diag/target/release/volta-server'}
assert cfg['binary'] in allowed
assert os.readlink('/proc/'+str(cfg['pid'])+'/exe')==cfg['binary']
os.kill(cfg['pid'],signal.SIGTERM)
for _ in range(150):
 raw=subprocess.check_output(['nvidia-smi','--query-gpu=memory.used','--format=csv,noheader,nounits'],text=True)
 if all(int(x.strip())<128 for x in raw.splitlines()):break
 time.sleep(.2)
else:raise RuntimeError('GPU memory did not release; no teacher started')
env=os.environ.copy()
for k in list(env):
 if k.startswith('VOLTA_'):env.pop(k)
env['LD_LIBRARY_PATH']='/usr/local/cuda-12.9/targets/x86_64-linux/lib:'+env.get('LD_LIBRARY_PATH','')
env['VOLTA_GRAPHS']='4'
if @@HEADS@@:env['VOLTA_HEADS']='1'
binary='/root/volta/target/release/volta'
digest=hashlib.file_digest(open(binary,'rb'),'sha256').hexdigest()
assert digest=='e216e64da3524d36b657f0cdc0cf3bf4a4099f421c5282970b908ead5817351c'
name=@@NAME@@
stdout='/root/teacher-'+name+'.jsonl';stderr='/root/teacher-'+name+'.log'
cmd=[binary,'eval','--model','/root/model/DeepSeek-V4.1-Flash-1017-Volta.gguf','--eval','/root/eval','--seqs','8','--len','1024','--chunk','1']
p=subprocess.Popen(cmd,env=env,stdin=subprocess.DEVNULL,stdout=open(stdout,'x'),stderr=open(stderr,'x'),start_new_session=True)
print(json.dumps({'name':name,'pid':p.pid,'heads':@@HEADS@@,'started_epoch':time.time(),'binary_sha256':digest,'stdout':stdout,'stderr':stderr,'command':cmd,'stopped_server_pid':cfg['pid']}))
'''.replace('@@HEADS@@', repr(a.heads)).replace('@@NAME@@', repr(a.name))
q = subprocess.run(SSH + ['python3 -'], input=code, text=True, capture_output=True)
(out / 'launch-stderr.txt').write_text(q.stderr)
assert q.returncode == 0, q.stderr
cfg = json.loads(q.stdout)
(out / 'configuration.json').write_text(json.dumps(cfg, indent=2))
print(json.dumps(cfg), flush=True)
deadline = min(time.time() + 600, 1792908206)
while time.time() < deadline:
    q = subprocess.run(SSH + ['cat ' + cfg['stdout']], text=True, capture_output=True)
    if q.returncode == 0 and q.stdout.strip():
        metric = json.loads(q.stdout.strip())
        assert metric['tokens'] == 8184
        baseline = json.loads((ROOT / 'receipts/teacher-eval.jsonl').read_text())
        metric['tight_gate_passed'] = (metric['ppl'] <= baseline['ppl'] + .1 and
            metric['kl_top256'] <= baseline['kl_top256'] + .005 and
            metric['top1_agree'] >= baseline['top1_agree'] - .005)
        (out / 'metrics.json').write_text(json.dumps(metric, indent=2))
        subprocess.run(['scp', '-F', str(ROOT / 'ssh.conf'),
                        'volta-run:' + cfg['stderr'], str(out / 'teacher.log')], check=True)
        assert metric['tight_gate_passed'], metric
        (out / 'TEACHER_PASSED').write_text(str(time.time()) + '\n')
        print(json.dumps(metric), flush=True)
        break
    observed = subprocess.run(SSH + ['ps -o stat= -p ' + str(cfg['pid'])], text=True, capture_output=True)
    if observed.returncode == 1 or (observed.returncode == 0 and observed.stdout.lstrip().startswith('Z')):
        log = subprocess.run(SSH + ['tail -60 ' + cfg['stderr']], text=True, capture_output=True)
        (out / 'failed-teacher.log').write_text(log.stdout + log.stderr)
        raise RuntimeError('teacher exited without result; inspect receipt')
    time.sleep(5)
else:
    raise TimeoutError('teacher observation ended; inspect same PID before any restart')
